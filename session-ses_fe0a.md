220: function startNetworkGraph() {
221:   const cvs = document.getElementById("network-canvas");
222:   if (!cvs) return;
223:   const ctx = cvs.getContext("2d");
224:   let width, height;
225: 
226:   function resize() {
227:     width = cvs.clientWidth;
228:     height = cvs.clientHeight;
229:     cvs.width = width;
230:     cvs.height = height;
231:   }
232:   window.addEventListener("resize", resize);
233:   resize();
234: 
235:   const nodes = Array.from({ length: 30 }).map(() => ({
236:     x: Math.random() * width,
237:     y: Math.random() * height,
238:     vx: (Math.random() - 0.5) * 0.5,
239:     vy: (Math.random() - 0.5) * 0.5,
240:   }));
241: 
242:   function draw() {
243:     ctx.clearRect(0, 0, width, height);
244:     ctx.lineWidth = 1;
245:     for (let i = 0; i < nodes.length; i++) {
246:       const n = nodes[i];
247:       n.x += n.vx;
248:       n.y += n.vy;
249:       if (n.x < 0 || n.x > width) n.vx *= -1;
250:       if (n.y < 0 || n.y > height) n.vy *= -1;
251: 
252:       for (let j = i + 1; j < nodes.length; j++) {
253:         const n2 = nodes[j];
254:         const dx = n.x - n2.x;
255:         const dy = n.y - n2.y;
256:         const dist = Math.sqrt(dx * dx + dy * dy);
257:         if (dist < 100) {
258:           ctx.beginPath();
259:           ctx.moveTo(n.x, n.y);
260:           ctx.lineTo(n2.x, n2.y);
261:           ctx.strokeStyle = `rgba(0, 240, 255, ${1 - dist / 100})`;
262:           ctx.stroke();
263:         }
264:       }
265: 
266:       ctx.beginPath();
267:       ctx.arc(n.x, n.y, 2, 0, Math.PI * 2);
268:       ctx.fillStyle = "#00f0ff";
269:       ctx.fill();
270:     }
271:     requestAnimationFrame(draw);
272:   }
273:   draw();
274: }
